K1    main         sum,mean,min,max:square,circle:4           SUMMATION=all WINDOW_FRAMES=4
# K2 reach beyond the 8 neighbouring tiles: r = 20 > T = 16 (halo across 5 x 5 tiles)
K2    halo         sum,mean,min,max:square,circle:20          SUMMATION=all
# K3 sliding windows: 2 frames per window, advancing by 1 (each frame in 2 windows, every tile emitted twice)
K3    sub4         sum,mean,min,max:square,circle:1,4         WINDOW_FRAMES=2 WINDOW_SLIDE=1
# K5 fractional data: S2 reflectance (DN * 1e-4 - 0.1) and synthetic normal values, every summation mode
K5    refl4,syn-normal  sum,mean:square:1,4,16                SUMMATION=all
K5    refl4,syn-normal  sum,mean:circle:4                     SUMMATION=exact
# K6 adversarial dynamic range: exact and Kahan within tolerance; plain running sums are expected to exceed it
K6    syn-adv      sum,mean:square:2,8                        SUMMATION=exact,kahan
K6    syn-adv      sum,mean:square:2,8                        SUMMATION=running EXPECT=fail
# K7 NoData: sparse NoData, blocks larger than the window (all-NoData windows), blocks across seams, a NoData tile
K7    nan4,syn-nodata  sum,mean,min,max:square,circle:1,4,16
# K8 missing and late tiles (deadline path): single tile, seam pair, tile row, raster-edge column; 1 and 4 frames per

# passed) and must be dropped: the result equals the gaps4 reference
K8    gaps4        sum,mean,min,max:square,circle:1,4         WINDOW_FRAMES=1,4
K8    gaps4        mean,max:square,circle:1,4                 NES_INPUT=gaps4-late NES_THREADS=1 CASE_TAG=late
# K9 thread stress: many small tiles, 16 worker threads (same reference as with 4)
K9    halo4        mean,max:square,circle:1,4,20              NES_THREADS=16 CASE_TAG=t16
# K11 arbitrary tile order: NES reads the frames with their maplets shuffled; reference, import and compare use the
# row-major store, so the results must match K1 / K2 (bit-identical; arrival order only changes when outputs leave)
K11   sub4         sum,mean,min,max:square,circle:1,4         SUMMATION=all NES_INPUT=shuf4 CASE_TAG=shuf
K11   sub4         mean,max:square:4                          NES_INPUT=shuf4 CASE_TAG=shuf WINDOW_FRAMES=4
K11   halo4        mean,max:square,circle:1,20                NES_INPUT=halo4-shuf CASE_TAG=shuf
