# Stores of the evaluation, built on demand by scripts/data/stores.sh (and by the suites, for the stores their cases use).
# name        recipe
#   s2 FRAMES TILE        first FRAMES non-empty Sentinel-2 scenes (0 = all), tile size TILE -> data/stores/<S2_NAME>-<FRAMES>[-t<TILE>]
#   derive BASE ARGS...   mabench derive of another store             -> data/stores/<S2_NAME>-<name> (S2-based) or <name>
#   synth ARGS...         mabench synth                               -> data/stores/<name>
# $VARIABLES are taken from bench.env.
main          s2 $S2_FRAMES $S2_TILE
halo          s2 $S2_FRAMES $S2_HALO_TILE
perf          s2 $S2_PERF_FRAMES $S2_TILE
sub2          derive perf --frames 0:2
sub4          derive perf --frames 0:4
sub8          derive perf --frames 0:8
halo4         derive halo --frames 0:4
# tile-size series on the same 4 scenes (G3): halo4 = 16, t32 ... t128, sub4 = 256, t512, t1024
t32           s2 4 32
t64           s2 4 64
t128          s2 4 128
t512          s2 4 512
t1024         s2 4 1024
# correctness inputs (K3-K11)
refl4         derive sub4 --scale 0.0001 --offset -0.1
nan4          derive sub4 --nan *:fraction=0.01 --nan 0:block=700,500,600,400 --nan 1:block=1000,1010,40,60 --nan 2:tile=5,5 --nan 3:block=1530,1530,30,30 --seed 1
gaps4         derive sub4 --drop-tiles 0:3,3 --drop-tiles 1:4,4 --drop-tiles 1:5,4 --drop-tiles 2:row=6 --drop-tiles 3:col=0
gaps4-late    derive sub4 --drop-tiles 1:4,4 --drop-tiles 1:5,4 --drop-tiles 2:row=6 --drop-tiles 3:col=0 --late-tiles 0:3,3@3
shuf4         derive sub4 --shuffle-tiles 1
halo4-shuf    derive halo4 --shuffle-tiles 1
syn-normal    synth --width 1024 --height 768 --generator normal --bands 2 --frames 4 --frame-period-ms 1000 --tile-size 128 --seed 1
syn-adv       synth --width 1024 --height 768 --generator normal --bands 1 --frames 2 --frame-period-ms 1000 --tile-size 128 --seed 2 --param outlier_fraction=0.001 --param outlier_scale=1e12
syn-nodata    synth --width 1024 --height 768 --generator waves --bands 1 --frames 4 --frame-period-ms 1000 --tile-size 128 --seed 3 --param nodata_fraction=0.05 --param nodata_block=100,100,300,400
